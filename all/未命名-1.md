```mermaid
graph TD
    %% 状态机主流程
    START((开始)) --> INIT[初始化: LCD, 密码加载, 电机归位, 红外];
    INIT --> S0[state=0: 显示输入提示];

    
    S3 --> POST_KEY{读取按键};
    
    POST_KEY -- NO --> S3;
    
    POST_KEY -- *(10) --> S3_CLOSE[关锁: 电机, 播放音乐];
    S3_CLOSE --> S0;
    
    POST_KEY -- #(12) --> S4[state=4: 密码设置模式 (进入子循环)];

    %% 状态 4: 密码设置
    S4 --> S4_USER[输入用户编号];
    S4_USER --> S4_PASS1[输入新密码 (第一次)];
    S4_PASS1 --> S4_PASS2[输入新密码 (第二次)];

    S4_PASS2 --> S4_CHECK{两次密码是否一致?};
    
    S4_CHECK -- 不一致 --> S4_FAIL[显示设置失败];
    S4_FAIL --> S5[state=5: 退出设置]; %% 退出设置，程序回到S3显示

    S4_CHECK -- 一致 --> S4_SAVE[保存密码 SavePassword()];
    S4_SAVE --> S5;
    
    S5 --> S3; %% 回到开箱后状态的显示

    %% 状态 8: 锁定状态 (手动)
    S8 --> S8_WAIT{按键等待};
    
    S8_WAIT -- NO --> S8;
    S8_WAIT -- UNLOCK(13) --> S0;
```
