```mermaid
graph TD
    %% 状态机主流程
    START((开始)) --> INIT[初始化: LCD, 密码加载, 电机归位, 红外];
    INIT --> S0[state=0: 显示输入提示];

    %% 状态 1: 密码输入与验证
    S0 --> S1[state=1: 输入等待 (*=确认, #=删除)];
    
    S1 --> INPUT{读取键盘/红外按键};
    
    INPUT -- 数字/0 --> DIGIT[记录数字, LCD显示*];
    DIGIT --> S1;

    INPUT -- #(12) --> DELETE[删除上一位, writestate--];
    DELETE --> S1;

    INPUT -- LOCK(13) --> S8[state=8: 锁定状态 (手动上锁)];
    
    INPUT -- *(10) --> VERIFY{调用 meach() 验证密码};
    
    VERIFY -- 匹配成功 (ok != 10) --> S2_SUCCESS[error=0, state=2: 验证通过];
    
    VERIFY -- 匹配失败 (ok == 10) --> ERROR_COUNT[error++];
    
    ERROR_COUNT --> CHECK_ERRORS{error == 5?};
    
    CHECK_ERRORS -- NO --> S0;

    CHECK_ERRORS -- YES --> LOCK_60s[调用 Lock60s() 锁定60秒];
    LOCK_60s --> S0;
```
